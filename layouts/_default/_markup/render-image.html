{{/* Responsive image rendering using a width-descriptor srcset (similar to Next.js' image optimization approach) rather than sources tied to hand-picked media-query breakpoints. The browser picks the best candidate from the srcset based on the "sizes" hint and the viewport's actual layout width/DPR, instead of us guessing exact breakpoints. */}} {{ $resource := .Page.Resources.GetMatch .Destination }} {{ if $resource }} {{/* Candidate widths, modeled after Next.js' default deviceSizes. */}} {{ $widths := slice 640 750 828 1080 1200 1920 }} {{ $candidates := slice }} {{ range $w := $widths }} {{ if le $w $resource.Width }} {{ $candidates = $candidates | append $w }} {{ end }} {{ end }} {{/* Small source images: fall back to the image's native width. */}} {{ if eq (len $candidates) 0 }} {{ $candidates = slice $resource.Width }} {{ end }} {{ $srcsetParts := slice }} {{ range $w := $candidates }} {{ $resized := $resource.Resize (printf "%dx" $w) }} {{ $srcsetParts = $srcsetParts | append (printf "%s %dw" $resized.RelPermalink $w) }} {{ end }} {{/* Use the largest candidate as the non-srcset-supporting fallback. */}} {{ $fallbackWidth := index $candidates (sub (len $candidates) 1) }} {{ $fallback := $resource.Resize (printf "%dx" $fallbackWidth) }} {{ else }} {{ end }}
